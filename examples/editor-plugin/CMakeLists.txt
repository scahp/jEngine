cmake_minimum_required(VERSION 3.24)
project(jEngineNativePlugin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JENGINE_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "jEngine repository root")
option(JENGINE_BUILD_PLUGIN_FIXTURES "Build real shared libraries for loader regression tests" OFF)
function(jengine_add_plugin target)
  add_library(${target} SHARED plugin.cpp)
  target_include_directories(${target} PRIVATE "${JENGINE_SOURCE_ROOT}/jEngine/Code/Engine/Editor")
  set_target_properties(${target} PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endfunction()
jengine_add_plugin(jengine_example_plugin)
if(JENGINE_BUILD_PLUGIN_FIXTURES)
  foreach(kind IN ITEMS v1 v2 badabi failload failrestore oversized missingexport)
    jengine_add_plugin(jengine_plugin_fixture_${kind})
    target_compile_definitions(jengine_plugin_fixture_${kind} PRIVATE JENGINE_PLUGIN_FIXTURE=1)
  endforeach()
  target_compile_definitions(jengine_plugin_fixture_v2 PRIVATE JENGINE_PLUGIN_EXAMPLE_VERSION=2)
  target_compile_definitions(jengine_plugin_fixture_badabi PRIVATE JENGINE_PLUGIN_BAD_ABI=1)
  target_compile_definitions(jengine_plugin_fixture_failload PRIVATE JENGINE_PLUGIN_FAIL_LOAD=1)
  target_compile_definitions(jengine_plugin_fixture_failrestore PRIVATE JENGINE_PLUGIN_FAIL_RESTORE=1)
  target_compile_definitions(jengine_plugin_fixture_oversized PRIVATE JENGINE_PLUGIN_OVERSIZED_STATE=1)
  target_compile_definitions(jengine_plugin_fixture_missingexport PRIVATE JENGINE_PLUGIN_MISSING_EXPORT=1)
endif()
